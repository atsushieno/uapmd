73 // Which parts of a captured track are applied when it is attached.
74 //
75 // Track duplication is not all-or-nothing in any DAW that offers it: the
76 // common choices are a full clone, and a track set up the same way but
77 // empty. Capture always takes everything, so one fragment serves a clone, a
78 // partial duplicate and an undo restore without being recaptured.
79 struct ProjectTrackAttachOptions {
80 ProjectObjectIdPolicy idPolicy{ProjectObjectIdPolicy::Mint};
81 // Negative appends. Undo restoration supplies the removed track's
82 // former index so project ordering is restored as well as identity.
83 int32_t insertionIndex{-1};
84 bool includePlugins{true};
85 // Instantiate the plugins but leave them at their defaults. Skipping
86 // state also skips the slowest part of attaching.
87 bool includePluginState{true};
88 bool includeClips{true};
89 };
