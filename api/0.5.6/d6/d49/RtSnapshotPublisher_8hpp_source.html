13 // Publishes immutable snapshots to a fixed number of real-time readers.
14 // Readers only perform lock-free pointer atomics. Snapshot allocation,
15 // retirement, reclamation, and destruction remain on the publishing thread.
16 //
17 // Each concurrently active reader must use a distinct reader index. Reader
18 // indices are fixed so entering the real-time path never registers a thread,
19 // allocates memory, or acquires a lock.
20 template<typename T, size_t ReaderCount = 1>
21 class RtSnapshotPublisher {
22 static_assert(ReaderCount > 0);
23 static_assert(std::atomic<const T*>::is_always_lock_free,
24 "RtSnapshotPublisher requires lock-free pointer atomics");

93 [[nodiscard]] Guard protect(size_t readerIndex = 0) const noexcept {
94 if (readerIndex >= ReaderCount)
95 return {};
96
97 const T* snapshot;
98 do {
99 snapshot = current_.load(std::memory_order_seq_cst);
100 hazards_[readerIndex].store(snapshot, std::memory_order_seq_cst);
101 } while (snapshot != current_.load(std::memory_order_seq_cst));
102 return Guard(this, readerIndex, snapshot);
103 }

105 // Publishing thread only. The old snapshot is reclaimed here when no
106 // reader protects it; otherwise it remains retired until a later publish
107 // or explicit reclaimRetired() call on the publishing thread.
108 void publish(std::unique_ptr<const T> snapshot) {
109 retired_.push_back(std::move(current_owner_));
110 current_owner_ = std::move(snapshot);
111 current_.store(current_owner_.get(), std::memory_order_seq_cst);
112 reclaimRetired();
113 }
114
115 void reclaimRetired() {
116 const auto protectedByReader = [this](const std::unique_ptr<const T>& candidate) {
117 if (!candidate)
118 return false;
119 return std::any_of(hazards_.begin(), hazards_.end(), [&candidate](const auto& hazard) {
120 return hazard.load(std::memory_order_seq_cst) == candidate.get();
121 });
122 };
123 std::erase_if(retired_, [&protectedByReader](const auto& candidate) {
124 return !protectedByReader(candidate);
125 });
126 }

128 // Publishing thread only. This is intended for non-RT code that already
129 // serializes snapshot publication and therefore needs no hazard slot.
