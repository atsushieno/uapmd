12 // Converts between real-world seconds and quarter-note beats using a piecewise-constant
13 // tempo map, and exposes the time-signature-in-effect over each beat range.
14 //
15 // "Beat" here always means a quarter note, matching BPM (quarter notes per minute) and
16 // MIDI tick-resolution (ticks per quarter note) conventions used elsewhere in the codebase.
17 //
18 // A map with no tempo data converts at the default BPM rebuild() was given (120 by default).
19 class TempoMap {
20 public:

22 double timeSeconds{0.0};
23 double bpm{120.0};
24 };

27 MidiTimeSignatureChange signature{};
28 };
29 struct EffectiveSignature {
30 double startBeat{0.0};
31 double endBeat{std::numeric_limits<double>::infinity()};
32 uint8_t numerator{4};
33 uint8_t denominator{4};

36 void rebuild(const std::vector<TempoPoint>& tempoPoints,
37 const std::vector<TimeSignaturePoint>& timeSignaturePoints,
38 double defaultBpm = kDefaultBpm) {
39 tempoSegments_.clear();
40 effectiveSignatures_.clear();
41 defaultBpm_ = defaultBpm > 0.0 ? defaultBpm : kDefaultBpm;
42
43 if (tempoPoints.empty()) {
44 hasTempoData_ = false;
45 } else {
46 hasTempoData_ = true;
47
48 double currentBpm = tempoPoints.front().bpm > 0.0 ? tempoPoints.front().bpm : defaultBpm;
49 double lastTime = 0.0;
50 double accumulatedBeats = 0.0;
51
52 for (const auto& point : tempoPoints) {
53 double eventTime = std::max(0.0, point.timeSeconds);
54 if (eventTime > lastTime) {
55 const double bpmToUse = currentBpm > 0.0 ? currentBpm : defaultBpm;
56 tempoSegments_.push_back(TempoSegment{lastTime, eventTime, bpmToUse, accumulatedBeats});
57 accumulatedBeats += (eventTime - lastTime) * (bpmToUse / 60.0);
58 lastTime = eventTime;
59 }
60 if (point.bpm > 0.0)
61 currentBpm = point.bpm;
62 }
63
64 const double bpmToUse = currentBpm > 0.0 ? currentBpm : defaultBpm;
65 tempoSegments_.push_back(TempoSegment{
66 lastTime,
67 std::numeric_limits<double>::infinity(),
68 bpmToUse,
69 accumulatedBeats
70 });
71 }
72
73 for (const auto& point : timeSignaturePoints) {
74 EffectiveSignature sig;
75 sig.startBeat = secondsToBeats(std::max(0.0, point.timeSeconds));
76 sig.numerator = point.signature.numerator;
77 sig.denominator = point.signature.denominator;
78 effectiveSignatures_.push_back(sig);
79 }
80 std::sort(effectiveSignatures_.begin(), effectiveSignatures_.end(),
81 [](const EffectiveSignature& a, const EffectiveSignature& b) { return a.startBeat < b.startBeat; });
82
83 // Bar one starts at beat zero. Without a signature there, the bars before the first
84 // change would have no meter to be counted in.
85 if (effectiveSignatures_.empty() || effectiveSignatures_.front().startBeat > 1e-9) {
86 EffectiveSignature implicit;
87 implicit.startBeat = 0.0;
88 effectiveSignatures_.insert(effectiveSignatures_.begin(), implicit);
89 }
90
91 // A meta event restating the meter already in force does not start a new region. It
92 // is the region's start that sets where its bars fall, so honouring a restatement
93 // would re-phase the bar grid onto whatever beat that event happens to sit on --
94 // moving every later bar line, and every genuine meter change with it.
95 effectiveSignatures_.erase(
96 std::unique(effectiveSignatures_.begin(), effectiveSignatures_.end(),
97 [](const EffectiveSignature& a, const EffectiveSignature& b) {
98 return a.numerator == b.numerator && a.denominator == b.denominator;
99 }),
100 effectiveSignatures_.end());
101
102 for (size_t i = 0; i < effectiveSignatures_.size(); ++i) {
103 effectiveSignatures_[i].endBeat = (i + 1 < effectiveSignatures_.size())
104 ? effectiveSignatures_[i + 1].startBeat
105 : std::numeric_limits<double>::infinity();
106 }
107 }

119 double secondsToBeats(double seconds) const {
120 if (tempoSegments_.empty())
121 return std::max(0.0, seconds) * (defaultBpm_ / 60.0);
122
123 const double clampedSeconds = std::max(0.0, seconds);
124 for (const auto& segment : tempoSegments_) {
125 if (clampedSeconds < segment.endTime) {
126 const double bpm = segment.bpm > 0.0 ? segment.bpm : kDefaultBpm;
127 return segment.accumulatedBeats + (clampedSeconds - segment.startTime) * (bpm / 60.0);
128 }
129 }
130
131 const auto& last = tempoSegments_.back();
132 const double bpm = last.bpm > 0.0 ? last.bpm : kDefaultBpm;
133 return last.accumulatedBeats + (clampedSeconds - last.startTime) * (bpm / 60.0);
134 }
135
136 double beatsToSeconds(double beats) const {
137 if (tempoSegments_.empty())
138 return std::max(0.0, beats) * (60.0 / defaultBpm_);
139
140 const double clampedBeats = std::max(0.0, beats);
141 for (const auto& segment : tempoSegments_) {
142 const double bpm = segment.bpm > 0.0 ? segment.bpm : kDefaultBpm;
143 double segmentEndBeats = std::numeric_limits<double>::infinity();
144 if (std::isfinite(segment.endTime))
145 segmentEndBeats = segment.accumulatedBeats + (segment.endTime - segment.startTime) * (bpm / 60.0);
146
147 if (clampedBeats < segmentEndBeats)
148 return segment.startTime + (clampedBeats - segment.accumulatedBeats) * (60.0 / bpm);
149 }
150
151 const auto& last = tempoSegments_.back();
152 const double bpm = last.bpm > 0.0 ? last.bpm : kDefaultBpm;
153 return last.startTime + (clampedBeats - last.accumulatedBeats) * (60.0 / bpm);
154 }

158 // How many quarter notes one bar of this meter spans (e.g. 3.5 for 7/8).
159 static double barLengthBeats(const EffectiveSignature& signature) {
160 const double numerator = signature.numerator > 0 ? signature.numerator : 4;
161 const double denominator = signature.denominator > 0 ? signature.denominator : 4;
162 return numerator * 4.0 / denominator;
163 }
