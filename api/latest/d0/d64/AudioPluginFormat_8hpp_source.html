15 // Which plugin operations have to happen on the UI/main thread. [flags]
16 //
17 // VST3 and CLAP require almost every plugin operation on the main thread, while AU and
18 // LV2 do not. This costs the application a lot of parallelism, so a format may report a
19 // narrower requirement, and may report it per plugin when only some of its plugins
20 // misbehave.
21 enum AudioPluginUIThreadRequirement : uint32_t {
22 UIThreadNotRequired = 0,
23 UIThreadForInstanceControl = 1,
24 UIThreadForParameters = 2,
25 UIThreadForState = 4,
26 UIThreadForAllNonAudioOperation = 0xFFFFFFFF
27 };

38 // A plugin format that this host can scan and instantiate.
39 //
40 // This is the abstraction the rest of uapmd-plugin-hosting is written against, and it is
41 // what an application implements to add a plugin format of its own (JSFX, SFZ, and so
42 // on). The formats we ship are implemented over remidy, but nothing here depends on
43 // remidy, so an implementation does not have to go through it.
44 //
45 // Register an implementation with PluginScanTool::addFormat() or
46 // AudioPluginHostingAPI::addPluginFormat().
47 class AudioPluginFormat {
48 public:
49 virtual ~AudioPluginFormat() = default;
50
51 // The format name. It identifies the format everywhere: it is what
52 // AudioPluginCatalogEntry::format() holds, and what callers pass to
53 // AudioPluginHostingAPI::createPluginInstance(). It must be unique among the
54 // registered formats.
55 virtual std::string name() = 0;
56
57 // Which operations on this plugin have to run on the UI/main thread.
58 virtual AudioPluginUIThreadRequirement requiresUIThreadOn(AudioPluginCatalogEntry* entry) = 0;
59
60 // How this format discovers installed plugins. Returning null disables scanning,
61 // which only makes sense for a format whose plugin list comes from somewhere else.
62 virtual AudioPluginScanning* scanning() = 0;
63
64 // Asynchronously creates a plugin instance, already configured for `options` and
65 // ready to have startProcessing() called on it.
66 // `callback` must be invoked exactly once, either with a non-null instance and an
67 // empty error, or with a null instance and a non-empty error message.
68 virtual void createInstance(AudioPluginCatalogEntry* entry,
69 const AudioPluginInstantiationOptions& options,
70 std::function<void(std::unique_ptr<AudioPluginInstanceAPI> instance, std::string error)> callback) = 0;
71 };
