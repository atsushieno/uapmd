82// A clip the host is offering as the subject of a ClipCommand. The identifiers
83// are the ones the host's own clip APIs take, so an addin resolves the clip
84// through the engine rather than through anything carried here; the two flags
85// are only enough for a command to say up front which clips it applies to.
86struct ClipCommandTarget {
87 int32_t track_index{-1};
88 int32_t clip_id{-1};
89 bool midi_clip{false};
90 bool master_track{false};
91};

93// Commands scoped to one clip rather than to the application. The host offers
94// these wherever it presents a clip, asking each command whether it applies to
95// that particular clip first. Addins must unregister during cleanup().
96class ClipCommand {
97public:
98 virtual ~ClipCommand() = default;
99
100 virtual std::string_view id() const noexcept = 0;
101 virtual std::string_view title() const noexcept = 0;
102 virtual int order() const noexcept { return 0; }
103 // False hides the command for this clip entirely; true but !enabled()
104 // shows it greyed out.
105 virtual bool appliesTo(const ClipCommandTarget& target) const noexcept = 0;
106 virtual bool enabled(const ClipCommandTarget& target) const noexcept { (void) target; return true; }
107 virtual void invoke(const ClipCommandTarget& target) noexcept = 0;
108};
