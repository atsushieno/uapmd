25 // One intended document change, expressed as a value.
26 //
27 // A command carries its whole payload -- the identities it addresses and
28 // the values it writes -- as ordinary data members. It must NOT carry
29 // std::function hooks that re-implement the mutation: that is what makes
30 // an operation unreadable, unloggable, and impossible to construct
31 // anywhere except at the one call site that knows how to bind it.
32 //
33 // A command binds to the *service* that owns the objects it mutates (a
34 // reference held from construction, supplied by whoever creates it). That
35 // is a service locator, not a re-implementation, and it keeps concrete
36 // commands in the module that owns their targets while this interface
37 // stays free of engine dependencies.
38 class ProjectCommand {
39 protected:
40 ProjectCommand() = default;
41
42 public:
43 virtual ~ProjectCommand() = default;
44
45 ProjectCommand(const ProjectCommand&) = delete;
46 ProjectCommand& operator=(const ProjectCommand&) = delete;
47
48 // Stable dotted identifier, for example "clip.setGain". Distinct from
49 // description(): the id is machine-facing (logging, coalescing, and
50 // eventually name-based dispatch from JavaScript and MCP), while the
51 // description is the user-visible history label.
52 virtual std::string_view commandId() const = 0;
53 virtual std::string description() const = 0;
54
55 // Bytes retained for as long as this command sits in history.
56 virtual size_t retainedSizeInBytes() const = 0;
57
58 // Coalescing of adjacent commands inside a gesture scope. Only ever
59 // called with a command of the same commandId(), so implementations
60 // may static_cast rather than dynamic_cast. Return true after adopting
61 // `subsequent`'s payload, so that a drag records one history step
62 // holding the initial and final values.
63 virtual bool mergeWith(const ProjectCommand& subsequent) {
64 return false;
65 }
66
67 // Whether the manager should wrap this execution in a document
68 // transaction, so that observers see one batch of events.
69 //
70 // Commands that finish asynchronously across external work -- a
71 // plug-in state request, an ARA archive -- must return false. A
72 // document transaction must not be held open across such a gap: an
73 // ARA plug-in cannot be archived while its document is being edited,
74 // which is exactly what an open transaction represents. Those commands
75 // batch their own events around the synchronous parts instead.
76 virtual bool batchesDocumentEvents() const {
77 return true;
78 }
79
80 // Performs the change. Implementations must invoke `completion`
81 // exactly once; synchronous commands do so before returning.
82 //
83 // Before mutating, an implementation captures the state it is about to
84 // overwrite and hands the manager the command that restores it, via
85 // ProjectCommandContext::recordRevert(). A command that finds the
86 // document already in the requested state simply completes without
87 // recording a revert, and no history entry is created -- which
88 // replaces the `if (before == after) return true;` early-out that
89 // otherwise has to be written into every mutation entry point.
90 virtual void execute(
91 ProjectCommandContext& context,
92 ProjectCommandCompletion completion) = 0;
93 };

108 virtual ProjectMutationOrigin origin() const = 0;
109
110 // True while this command is running as an undo or redo of an earlier
111 // step. Commands rarely need it -- the manager already suppresses
112 // history recording during replay -- but a command that drives
113 // external state (a plug-in editor, an ARA document) may want to
114 // distinguish a fresh edit from a replay.
115 virtual bool replaying() const = 0;
116
117 // Registers the command that reverses what this command is about to
118 // do, built from state captured before mutating.
119 //
120 // The reverting command is expected to register the original command
121 // as its own revert when it runs. That symmetry is what makes redo
122 // work without a separate code path: undo executes the revert, which
123 // re-registers the original, which redo then executes.
124 //
125 // Calling this more than once within a single execute() replaces the
126 // previous registration; not calling it at all means "this execution
127 // changed nothing".
128 virtual void recordRevert(ProjectCommandPtr revert) = 0;
129
130 // Marshals `task` onto the model thread. Asynchronous commands finish
131 // on whichever thread completes their external work (a plug-in state
132 // request, a file read); every document mutation and every call to
133 // recordRevert() must be dispatched through here first.
134 virtual void dispatchToModelThread(ProjectUndoTask task) = 0;
135 };
