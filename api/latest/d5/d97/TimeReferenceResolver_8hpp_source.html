34 // A marker is unique only within the object that owns it, so addressing one
35 // takes both the owner's reference identifier and the marker's own.
36 struct MarkerKey {
37 std::string clipReferenceId;
38 std::string markerId;

40 bool operator==(const MarkerKey& other) const {

47 size_t seed = std::hash<std::string>{}(key.clipReferenceId);
48 seed ^= std::hash<std::string>{}(key.markerId) + 0x9e3779b97f4a7c15ull + (seed << 6) + (seed >> 2);
49 return seed;
