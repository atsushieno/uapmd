9 // Describes one plugin that some AudioPluginFormat can instantiate.
10 //
11 // It is the uapmd_plugin_hosting counterpart of remidy::PluginCatalogEntry, and it is a
12 // plain value type: everything here is what a scanner discovered about the plugin, and
13 // the format is expected to be able to instantiate the plugin from `format` + `pluginId`
14 // alone (plus `bundlePath` for bundle-based formats).
15 class AudioPluginCatalogEntry {

20 std::string product_url_{};

23 public:
24 // The format name, matching AudioPluginFormat::name().

29 void pluginId(const std::string& value) { plugin_id_ = value; }

34 const std::string& productUrl() const { return product_url_; }
35 void productUrl(const std::string& value) { product_url_ = value; }
36 // A file system path, or a URL for formats whose plugins are remote (WebCLAP).
37 // It is empty for formats that have no bundle concept (AU).
