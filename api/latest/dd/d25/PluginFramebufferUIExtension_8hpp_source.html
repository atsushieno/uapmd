10// A plugin editor that is a pixel buffer rather than a native window.
11//
12// The UI members of AudioPluginInstanceAPI describe a native view: the host passes a parent
13// window handle and the plugin puts a view inside it. Some formats have no view to give.
14// JSFX draws into a buffer the host supplies, and the host is then responsible for putting
15// those pixels on screen and for sending input back.
16//
17// A format whose editor works that way registers this extension. The host draws the
18// framebuffer wherever it likes -- for an immediate mode GUI that means uploading it as a
19// texture each time the frame serial changes -- and reports input through deliverInput().
20//
21// hasUISupport() still reports whether there is an editor at all. A format that registers
22// this extension reports true there and leaves createUI()/showUI() to the base class
23// defaults, because there is no native view to create.
24namespace uapmd_plugin_hosting {
25 inline constexpr std::string_view kPluginFramebufferUIExtensionId =
26 "dev.atsushieno.uapmd.plugin-instance.framebuffer-ui.v1";

28 // A view of the most recently rendered frame. It is only valid for the duration of the
29 // PluginFramebufferUI::readFrame() callback that produced it.
30 struct FramebufferView {
31 const uint8_t* pixels{nullptr};
32 uint32_t width{0};
33 uint32_t height{0};
34 // Distance in bytes between the start of one row and the next.
35 uint32_t stride{0};
36 // Increases every time the plugin draws a new frame. A host that uploads the pixels
37 // to a texture compares this against what it uploaded last, and skips the upload
38 // when nothing has changed.
39 uint64_t serial{0};
40 };

57 // Everything the plugin should know about input since the last call. The host fills it
58 // in each time it has something to report; it is a snapshot for the pointer state and a
59 // queue for keys, because a pointer only has a current position while key presses must
60 // not be dropped.
61 struct FramebufferInput {
62 // Pointer position in framebuffer pixels, with the origin at the top left.
63 int32_t pointerX{0};
64 int32_t pointerY{0};
65 // Bitmask of the buttons currently held.
66 uint32_t buttons{0};
67 // Bitmask of the modifier keys currently held.
68 uint32_t modifiers{0};
69 // Scroll since the last call, in steps normalised to +/-1.0.
70 double wheel{0.0};
71 double horizontalWheel{0.0};
72 bool hasFocus{false};
73 bool visible{true};
74 bool pointerOver{false};
75 std::vector<FramebufferKeyEvent> keys{};
76 };

78 // Services the plugin needs from whoever is displaying it.
79 //
80 // None of these may block. A plugin format whose script asks for a menu has to wait for
81 // the answer -- that is inherent to what a menu is -- but it waits on its own rendering
82 // thread, never on the host's. So the host is asked to *open* a menu and report the
83 // outcome later, which it can do from its own event loop without a modal loop and
84 // without blocking anything.
85 //
86 // Putting the waiting on the plugin's side rather than the host's is deliberate: it is
87 // the part that deadlocks if it is got wrong, and there is one implementation of it
88 // rather than one per host.
89 class FramebufferUIHost {
90 public:
91 virtual ~FramebufferUIHost() = default;
92
93 // Opens a menu and returns immediately. `spec` is the format's own menu
94 // description and the position is in framebuffer pixels.
95 //
96 // `completed` must be called exactly once, with the chosen item counting from 1,
97 // or 0 if the user dismissed the menu without choosing. It may be called from any
98 // thread, including from inside this call. A host that cannot show a menu at all
99 // still has to complete with 0 rather than dropping the request.
100 virtual void requestMenu(const std::string& spec, int32_t x, int32_t y,
101 std::function<void(int32_t)> completed) = 0;
102
103 // Requests a mouse cursor shape, using the format's own cursor identifiers.
104 virtual void setCursor(int32_t cursor) = 0;
105
106 // Returns the path of a file dropped on the plugin, by index, or an empty string
107 // when there is none. An index of -1 asks the host to forget the dropped files.
108 virtual std::string droppedFile(int32_t index) = 0;
109 };

119 // The size the plugin would like when it is first shown, which the host may use to
120 // size its panel. It is a hint and nothing more: after the first frame the host's
121 // size decides, and the plugin draws to whatever surfaceSize() it was last given.
122 // Returns false when the plugin has no preference.
123 virtual bool preferredSize(uint32_t& width, uint32_t& height) = 0;
124
125 // Sets the surface the plugin draws into. `scaleFactor` is the display scale, 1.0
126 // for a conventional display and 2.0 for a doubled one; a plugin that does not ask
127 // for high resolution output ignores it. Width and height are in framebuffer
128 // pixels and already include the scale.
129 //
130 // Call it before the first readFrame() and whenever the panel changes size. The
131 // change takes effect on the plugin's next frame, so a readFrame() immediately
132 // afterwards can still return the previous size -- hosts should use the width and
133 // height from FramebufferView rather than the ones they last set.
134 //
135 // on: UI thread
136 virtual void surfaceSize(uint32_t width, uint32_t height, double scaleFactor) = 0;
137
138 // Calls `consume` with the current frame, holding the plugin's frame lock for the
139 // duration, and returns whether there was a frame to give. `consume` must not call
140 // back into the plugin, and must not keep the pointer.
141 //
142 // on: UI thread
143 virtual bool readFrame(const std::function<void(const FramebufferView&)>& consume) = 0;
144
145 // Hands input to the plugin. Key events are consumed; pointer and window state
146 // replace whatever was reported before.
147 //
148 // on: UI thread
149 virtual void deliverInput(FramebufferInput input) = 0;
150
151 // Sets who services requestMenu(), setCursor() and droppedFile(). Passing nullptr
152 // detaches and abandons any menu still open, releasing whatever was waiting on it.
153 // The host must keep the object alive until it has detached.
154 //
155 // Detaching is not required before destroying the plugin: an implementation has to
156 // release its own waiters on the way down regardless, because the alternative is a
157 // teardown that hangs.
158 //
159 // on: UI thread
160 virtual void uiHost(FramebufferUIHost* host) = 0;
161
162 // Tells the plugin whether anyone is looking. A plugin that is not being displayed
163 // need not draw, and hosts should say so rather than discarding frames.
164 //
165 // on: UI thread
166 virtual void displayed(bool value) = 0;
167 };
